# 경쟁 모드 롤백 · 팀(클랜) 탭 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 경쟁 모드를 코드·DB에서 모두 걷어내고, 팀 현황 페이지에 결과 등록 참여자를 팀(클랜)으로 묶는 트로피 탭과 클랜 통계 배너(빈 화면으로 이동)를 추가한다.

**Architecture:** 롤백은 경쟁 모드 도입 직전 커밋(`ce284f3^`)의 파일을 되살리는 방식으로 하고, 경쟁 모드가 아닌 커밋이 섞인 4개 파일(`server/index.js`, `src/App.jsx`, `src/utils/calculateAssets.js`, `src/pages/ParentGuide.jsx`)만 손으로 맞춘다. 클랜은 서버 메모리 모듈(`server/clans.js`)에 두고, 참여자 목록은 DB의 결과 행(`getRankings`)을 읽어 순수 함수(`server/clanView.js`)로 합친다. 화면은 `AdminClassDashboard`에 트로피 탭을 더하고, 탭 본문은 `ClanTab` 컴포넌트가 맡는다.

**Tech Stack:** React 18 + Vite, CSS Modules, Express + socket.io, Supabase(Postgres), Vitest + Testing Library

**설계 문서:** `docs/superpowers/specs/2026-10-29-clan-tab-design.md`

**설계와 다른 점(구현 단순화):**
- 결과 등록·등록된 팀 삭제 후 서버가 `class-clans-updated`를 따로 보내지 않는다. 이 경우 이미 `class-rooms-updated`가 나가므로, 트로피 탭이 두 이벤트를 모두 듣고 클랜 목록을 다시 불러온다. 결과는 같다.
- 참여자의 팀(게임) 이름은 기존 `mapRankingRow`의 `teamName`(세션 제목, 없으면 팀 코드)을 그대로 쓴다.

**테스트 실행:** `npx vitest run <경로>` (전체는 `npx vitest run`)

```js
import { describe, it, expect, beforeEach } from 'vitest'
import {
  createClan, renameClan, deleteClan, assignToClan, listClans, pruneMembers, deleteClassClans,
  isValidClanName, _resetClans,

beforeEach(() => _resetClans())

describe('clans', () => {
  it('수업별로 "팀 N" 이름으로 만들고, 지워도 번호를 다시 쓰지 않는다', () => {
    const a = createClan('c1')

    expect([a.name, b.name]).toEqual(['팀 1', '팀 2'])
    deleteClan('c1', b.id)
    expect(createClan('c1').name).toBe('팀 3')
    expect(createClan('c2').name).toBe('팀 1')
  })

  it('만든 순서대로 목록을 돌려준다', () => {
    const a = createClan('c1')
    const b = createClan('c1')
    expect(listClans('c1').map(c => c.id)).toEqual([a.id, b.id])
    expect(listClans('none')).toEqual([])
  })

  it('이름 검증: 앞뒤 공백을 빼고 1~20자', () => {

    expect(isValidClanName('   ')).toBe(false)
    expect(isValidClanName('가'.repeat(21))).toBe(false)
    expect(isValidClanName(3)).toBe(false)
  })

  it('이름을 바꾼다(앞뒤 공백 제거). 없는 클랜이면 null', () => {
    const a = createClan('c1')
    expect(renameClan('c1', a.id, '  호랑이반 ').name).toBe('호랑이반')
    expect(listClans('c1')[0].name).toBe('호랑이반')
    expect(renameClan('c2', a.id, '다른 수업')).toBeNull()
  })

  it('배정·이동·해제한다. 한 참여자는 한 클랜에만 있다', () => {
    const a = createClan('c1')
    const b = createClan('c1')

    expect(assignToClan('c1', 'r1', b.id)).toBe(true)
    expect(listClans('c1').map(c => c.memberIds)).toEqual([[], ['r1']])

  it('없는 클랜으로는 배정하지 않는다', () => {
    const a = createClan('c1')

    expect(createClan('c1').name).toBe('팀 1')
  })
})
```

```js
export function isValidClanName(name) {
  if (typeof name !== 'string') return false
  const trimmed = name.trim()
  return trimmed.length >= 1 && trimmed.length <= MAX_CLAN_NAME_LENGTH
}

function entryOf(classId) {
  let entry = clansByClass.get(classId)
  if (!entry) {
    entry = { counter: 0, clans: new Map() }
    clansByClass.set(classId, entry)
  }
  return entry
}

function toPublic(clan) {
  return { id: clan.id, name: clan.name, memberIds: [...clan.memberIds] }
}

function findClan(classId, clanId) {
  return clansByClass.get(classId)?.clans.get(clanId) ?? null
}

export function createClan(classId) {
  const entry = entryOf(classId)
  entry.counter += 1
  const clan = { id: randomUUID(), name: `팀 ${entry.counter}`, memberIds: new Set() }
  entry.clans.set(clan.id, clan)
  return toPublic(clan)
}

// 이름 검증은 호출하는 쪽(isValidClanName)에서 먼저 한다.
export function renameClan(classId, clanId, name) {
  const clan = findClan(classId, clanId)
  if (!clan) return null
  clan.name = name.trim()
  return toPublic(clan)
}

export function deleteClan(classId, clanId) {
  return clansByClass.get(classId)?.clans.delete(clanId) ?? false
}

// clanId가 null이면 배정 해제. 다른 클랜에 있던 참여자는 그 클랜에서 빠진다.
export function assignToClan(classId, resultId, clanId) {
  const target = clanId === null ? null : findClan(classId, clanId)
  if (clanId !== null && !target) return false
  for (const clan of entryOf(classId).clans.values()) clan.memberIds.delete(resultId)
  target?.memberIds.add(resultId)
  return true
}

// Map은 넣은 순서를 지키므로 만든 순서대로 나온다.
export function listClans(classId) {
  const entry = clansByClass.get(classId)
  return entry ? [...entry.clans.values()].map(toPublic) : []
}

// 등록된 팀이 지워져 DB에 없는 결과 id를 정리한다.
export function pruneMembers(classId, validResultIds) {
  const entry = clansByClass.get(classId)
  if (!entry) return
  for (const clan of entry.clans.values()) {
    for (const id of clan.memberIds) {
      if (!validResultIds.has(id)) clan.memberIds.delete(id)
    }
  }
}

export function deleteClassClans(classId) {
  clansByClass.delete(classId)
}

export function _resetClans() {
  clansByClass.clear()
}
```

```js
function toParticipant(row) {

}

export function buildClanView(rows, clans) {
  const byId = new Map(rows.map(row => [row.resultId, toParticipant(row)]))
  const assigned = new Set()
  const clanViews = clans.map(clan => ({
    id: clan.id,
    name: clan.name,
    members: clan.memberIds
      .filter(id => byId.has(id))
      .map(id => {
        assigned.add(id)
        return byId.get(id)
      }),
  }))
  const participants = [...byId.values()]
    .filter(p => !assigned.has(p.resultId))
    .sort((a, b) => a.teamName.localeCompare(b.teamName, 'ko') || a.name.localeCompare(b.name, 'ko'))
  return { participants, clans: clanViews }
}
```

```js
// 트로피 탭의 낙관적 갱신: 서버 응답 전에 화면의 { participants, clans }에서 참여자 한 명을 옮긴다.
// toClanId가 null이면 참여자 블럭으로 돌려보낸다.
export function moveInView(view, resultId, toClanId) {
  const member = [...view.participants, ...view.clans.flatMap(c => c.members)]
    .find(m => m.resultId === resultId)
  if (!member) return view
  const without = list => list.filter(m => m.resultId !== resultId)
  return {
    participants: toClanId === null ? [...without(view.participants), member] : without(view.participants),
    clans: view.clans.map(clan => ({
      ...clan,
      members: clan.id === toClanId ? [...without(clan.members), member] : without(clan.members),
    })),
  }
}
```

```jsx
import { useState, useRef } from 'react'
import { CLAN_MEMBER_DRAG_TYPE } from '../../constants/clan'
import styles from './ClanCard.module.css'

const VISIBLE_SLOTS = 4
const MAX_NAME_LENGTH = 20

// 트로피 탭의 팀(클랜) 카드. 2×2 슬롯에 멤버를 보여주고, 5명 이상이면 "외 N명"으로 펼친다.
export default function ClanCard({ clan, onDropMember, onRename, onDelete }) {
  const [editing, setEditing] = useState(false)
  const [draft, setDraft] = useState(clan.name)
  const [expanded, setExpanded] = useState(false)
  const [isOver, setIsOver] = useState(false)
  // Enter·Esc로 끝낸 뒤 입력창이 사라지며 blur가 한 번 더 와도 다시 저장하지 않게 막는다.
  const finishedRef = useRef(false)

  const { members } = clan
  const hiddenCount = Math.max(0, members.length - VISIBLE_SLOTS)
  const visible = expanded ? members : members.slice(0, VISIBLE_SLOTS)
  const emptySlots = Math.max(0, VISIBLE_SLOTS - members.length)

  function startEditing() {
    finishedRef.current = false
    setDraft(clan.name)
    setEditing(true)
  }

  function commit() {
    if (finishedRef.current) return
    finishedRef.current = true
    setEditing(false)
    const name = draft.trim()
    if (!name || name.length > MAX_NAME_LENGTH || name === clan.name) return
    onRename(clan.id, name)
  }

  function handleKeyDown(e) {
    if (e.key === 'Enter') commit()
    if (e.key === 'Escape') {
      finishedRef.current = true
      setEditing(false)
    }
  }

  function handleDrop(e) {
    e.preventDefault()
    setIsOver(false)
    const resultId = e.dataTransfer.getData(CLAN_MEMBER_DRAG_TYPE)
    if (!resultId || members.some(mb => mb.resultId === resultId)) return
    onDropMember(resultId, clan.id)
  }

  return (
    <section
      className={`${styles.card} ${isOver ? styles.over : ''}`}
      aria-label={clan.name}
      onDragOver={e => { e.preventDefault(); setIsOver(true) }}
      onDragLeave={() => setIsOver(false)}
      onDrop={handleDrop}
    >
      <div className={styles.header}>
        {editing ? (
          <input
            className={styles.nameInput}
            aria-label="팀 이름"
            value={draft}
            maxLength={MAX_NAME_LENGTH}
            autoFocus
            onChange={e => setDraft(e.target.value)}
            onKeyDown={handleKeyDown}
            onBlur={commit}
          />
        ) : (
          <button type="button" className={styles.name} aria-label={`${clan.name} 이름 바꾸기`} onClick={startEditing}>
            {clan.name}
          </button>
        )}
        <span className={styles.count}>{members.length}명</span>
        <button type="button" className={styles.deleteBtn} aria-label={`${clan.name} 삭제`} onClick={() => onDelete(clan)}>
          ×
        </button>
      </div>

      <ul className={styles.slots}>
        {visible.map(mb => (
          <li
            key={mb.resultId}
            className={styles.member}
            draggable
            onDragStart={e => e.dataTransfer.setData(CLAN_MEMBER_DRAG_TYPE, mb.resultId)}
          >
            <img src={`/characters/${mb.character}.png`} alt="" className={styles.avatar} />
            <span className={styles.memberName}>{mb.name}</span>
            <span className={styles.memberTeam}>{mb.teamName}</span>
          </li>
        ))}
        {Array.from({ length: emptySlots }, (_, i) => (
          <li key={`empty-${i}`} className={styles.emptySlot} data-testid="empty-slot" />
        ))}
      </ul>

      {hiddenCount > 0 && (
        <button type="button" className={styles.moreBtn} onClick={() => setExpanded(v => !v)}>
          {expanded ? '접기' : `외 ${hiddenCount}명`}
        </button>
      )}
    </section>
  )
}
```

```jsx
import { useState, useEffect, useCallback } from 'react'
import ClanParticipantBlock from './ClanParticipantBlock'
import ClanCard from './ClanCard'
import ConfirmDialog from './ConfirmDialog'
import { adminFetch } from '../../utils/adminAuth'
import { toast } from '../../utils/toast'
import { moveInView } from '../../utils/clanView'
import { useSocketContext } from '../../contexts/SocketContext'
import styles from './ClanTab.module.css'

// 팀 현황 페이지의 트로피 탭 본문: 결과 등록 참여자를 팀(클랜)으로 묶는다.
// 클랜은 서버 메모리에 있으므로 서버가 재시작되면 비어서 내려온다.
export default function ClanTab({ classId }) {
  const { socket } = useSocketContext()
  const [view, setView] = useState(null)
  const [loadError, setLoadError] = useState(false)
  const [deleteTarget, setDeleteTarget] = useState(null)
  const base = `/api/admin/classes/${encodeURIComponent(classId)}/clans`

  const load = useCallback(() => {
    adminFetch(base)
      .then(r => { if (!r.ok) throw new Error(); return r.json() })
      .then(data => { setView(data); setLoadError(false) })
      .catch(() => setLoadError(true))
  }, [base])

  useEffect(() => { load() }, [load])

  // 클랜 변경은 class-clans-updated, 결과 등록·등록된 팀 삭제는 class-rooms-updated로 알 수 있다.
  // 채널 가입(watch-class-rooms)은 AdminClassDashboard가 한다.
  useEffect(() => {
    if (!socket) return
    socket.on('class-clans-updated', load)
    socket.on('class-rooms-updated', load)
    return () => {
      socket.off('class-clans-updated', load)
      socket.off('class-rooms-updated', load)
    }
  }, [socket, load])

  async function send(path, options, fallbackError) {
    try {
      const res = await adminFetch(`${base}${path}`, {
        ...options,
        headers: { 'Content-Type': 'application/json' },
      })
      if (res.ok) return true
      const { error } = await res.json().catch(() => ({}))
      toast(error || fallbackError)
    } catch {
      toast(fallbackError)
    }
    load()
    return false
  }

  function handleMove(resultId, toClanId) {
    setView(v => moveInView(v, resultId, toClanId))
    send('/assign', { method: 'POST', body: JSON.stringify({ resultId, toClanId }) }, '팀 배정에 실패했어요')
  }

  async function handleCreate() {
    if (await send('', { method: 'POST' }, '팀을 만들지 못했어요')) load()
  }

  function handleRename(clanId, name) {
    setView(v => ({ ...v, clans: v.clans.map(c => (c.id === clanId ? { ...c, name } : c)) }))
    send(`/${clanId}`, { method: 'PATCH', body: JSON.stringify({ name }) }, '팀 이름을 바꾸지 못했어요')
  }

  async function handleConfirmDelete() {
    const target = deleteTarget
    setDeleteTarget(null)
    if (await send(`/${target.id}`, { method: 'DELETE' }, '팀을 삭제하지 못했어요')) load()
  }

  if (loadError && !view) return <p className={styles.message}>팀 정보를 불러오지 못했어요.</p>
  if (!view) return <p className={styles.message}>불러오는 중...</p>

  return (
    <>
      <ClanParticipantBlock participants={view.participants} onDropMember={resultId => handleMove(resultId, null)} />
      <div className={styles.grid}>
        {view.clans.map(clan => (
          <ClanCard
            key={clan.id}
            clan={clan}
            onDropMember={handleMove}
            onRename={handleRename}
            onDelete={setDeleteTarget}
          />
        ))}
        <button type="button" className={styles.createCard} onClick={handleCreate}>
          <span className={styles.createIcon} aria-hidden="true">+</span>
          <span className={styles.createLabel}>팀 만들기</span>
        </button>
      </div>

      {deleteTarget && (
        <ConfirmDialog
          tone="danger"
          title="팀 삭제"
          description={<>{deleteTarget.name}을(를) 삭제할까요?<br />팀원은 참여자 목록으로 돌아갑니다.</>}
          onCancel={() => setDeleteTarget(null)}
          onConfirm={handleConfirmDelete}
        />
      )}
    </>
  )
}
```
